cmake_minimum_required(VERSION 3.24)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" OPENEDDI_VERSION LIMIT_COUNT 1)
project(OpenEDDI VERSION ${OPENEDDI_VERSION} LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Portable Windows packages must not require a separately installed VC runtime.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
# Optional offline corresponding-source bundle.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/deps/openbmh")
 foreach(dep openbmh sdl2 miniz lzma_sdk nfd rmlui freetype editor_json editor_http editor_images editor_svg editor_xml)
  string(TOUPPER "${dep}" upper)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/deps/${dep}")
   set(FETCHCONTENT_SOURCE_DIR_${upper} "${CMAKE_CURRENT_SOURCE_DIR}/deps/${dep}" CACHE PATH "Offline dependency" FORCE)
  endif()
 endforeach()
endif()

include(CTest)
set(OPENBMH_BUILD_EDITOR ON CACHE BOOL "Shared editor support" FORCE)
FetchContent_Declare(openbmh GIT_REPOSITORY https://github.com/Weyoune/openbmh.git GIT_TAG b879d5be008739f69afef562fb4bce56f1bdc778)
FetchContent_GetProperties(openbmh)
if(NOT openbmh_POPULATED)
  FetchContent_Populate(openbmh)
endif()
# The reused project uses this root for its staged assets and optional tests.
set(openeddi_root "${CMAKE_SOURCE_DIR}")
find_package(Python3 COMPONENTS Interpreter REQUIRED)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${openeddi_root}/tools/patch_dependency.py" "${openbmh_SOURCE_DIR}" COMMAND_ERROR_IS_FATAL ANY)
set(CMAKE_SOURCE_DIR "${openbmh_SOURCE_DIR}")
add_subdirectory("${openbmh_SOURCE_DIR}" "${openbmh_BINARY_DIR}" EXCLUDE_FROM_ALL)
set(CMAKE_SOURCE_DIR "${openeddi_root}")
FetchContent_GetProperties(editor_http)
find_package(Threads REQUIRED)
set(BMH_SDL_TARGET SDL2::SDL2-static)
if(WIN32)
 set(BMH_SDL_TARGET SDL2::SDL2main SDL2::SDL2-static)
endif()
add_library(openeddi_asset_io STATIC src/editor/archive.cpp src/editor/images.cpp)
target_include_directories(openeddi_asset_io PUBLIC src)
target_link_libraries(openeddi_asset_io PUBLIC nlohmann_json::nlohmann_json bmh_content PRIVATE bmh_miniz ${BMH_SDL_TARGET} SDL2_image::SDL2_image-static lunasvg::lunasvg pugixml::pugixml)
add_library(openeddi_core STATIC src/editor/world.cpp src/editor/session.cpp src/editor/comparison.cpp src/editor/regulations.cpp src/editor/options.cpp src/editor/mcp.cpp src/legacy/document.cpp src/legacy/transaction.cpp src/legacy/graphics.cpp)
target_include_directories(openeddi_core BEFORE PUBLIC src "${editor_http_SOURCE_DIR}")
target_compile_definitions(openeddi_core PUBLIC OPENBMH_EDITOR_VERSION="${OPENEDDI_VERSION}")
target_link_libraries(openeddi_core PUBLIC openeddi_asset_io nlohmann_json::nlohmann_json bmh_content Threads::Threads PRIVATE bmh_miniz ${BMH_SDL_TARGET} SDL2_image::SDL2_image-static lunasvg::lunasvg pugixml::pugixml)
if(WIN32)
 target_link_libraries(openeddi_core PRIVATE ws2_32 crypt32)
endif()
add_executable(OpenEDDI src/editor/main.cpp src/editor/ui.cpp)
target_link_libraries(OpenEDDI PRIVATE openeddi_core bmh_rmlui nfd::nfd ${BMH_SDL_TARGET})
if(WIN32)
 set_target_properties(OpenEDDI PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
if(MSVC)
 target_compile_options(openeddi_core PRIVATE /utf-8)
 target_compile_options(OpenEDDI PRIVATE /utf-8)
endif()
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/assets/editor")
foreach(asset editor.rml theme.rcss)
 configure_file("assets/editor/${asset}" "${CMAKE_BINARY_DIR}/assets/editor/${asset}" COPYONLY)
endforeach()
FetchContent_GetProperties(rmlui)
foreach(weight Regular Bold)
 configure_file("${rmlui_SOURCE_DIR}/Samples/assets/LatoLatin-${weight}.ttf" "${CMAKE_BINARY_DIR}/assets/editor/LatoLatin-${weight}.ttf" COPYONLY)
endforeach()
add_executable(legacy_tests tests/legacy_tests.cpp)
target_link_libraries(legacy_tests PRIVATE openeddi_core)
add_test(NAME openeddi_legacy COMMAND legacy_tests)
add_test(NAME openeddi_version COMMAND OpenEDDI --version)

foreach(test data graphics)
 add_executable(${test}_tests tests/${test}_tests.cpp)
 target_link_libraries(${test}_tests PRIVATE openeddi_core)
 add_test(NAME openeddi_${test} COMMAND ${test}_tests)
endforeach()

add_executable(transaction_failure_tests tests/transaction_failure_tests.cpp)
target_link_libraries(transaction_failure_tests PRIVATE openeddi_core)
add_test(NAME openeddi_transaction_failures COMMAND transaction_failure_tests)

if(MSVC)
 foreach(target openeddi_asset_io legacy_tests data_tests graphics_tests transaction_failure_tests)
  target_compile_options(${target} PRIVATE /utf-8)
 endforeach()
 target_compile_definitions(openeddi_core PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
add_executable(mcp_tests tests/mcp_tests.cpp)
target_link_libraries(mcp_tests PRIVATE openeddi_core)
add_test(NAME openeddi_mcp COMMAND mcp_tests)

add_executable(display_tests tests/display_tests.cpp)
target_include_directories(display_tests PRIVATE src)
add_test(NAME openeddi_display_metrics COMMAND display_tests)
